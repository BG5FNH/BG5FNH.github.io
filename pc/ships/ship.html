import * as THREE from 'three'; /** * 创建一艘最简单的太空飞船（轮廓风格） * 返回一个 THREE.Group，包含机身、机翼、尾翼和边缘线框 */ export function createShip() { const ship = new THREE.Group(); // ---- 材质定义 ---- const bodyMat = new THREE.MeshStandardMaterial({ color: 0xcccccc, roughness: 0.4, metalness: 0.6, }); const wingMat = new THREE.MeshStandardMaterial({ color: 0x888888, roughness: 0.5, metalness: 0.4, }); const engineMat = new THREE.MeshStandardMaterial({ color: 0x333333, roughness: 0.3, metalness: 0.8, }); // ---- 机身（圆柱体，沿Z轴） ---- const bodyGeom = new THREE.CylinderGeometry(0.8, 0.8, 4, 12); bodyGeom.rotateX(Math.PI / 2); // 让圆柱体沿着Z轴 const body = new THREE.Mesh(bodyGeom, bodyMat); ship.add(body); // ---- 机头（锥体，尖端指向-Z） ---- const noseGeom = new THREE.ConeGeometry(0.8, 2, 12); noseGeom.rotateX(-Math.PI / 2); // 尖端向前（-Z） const nose = new THREE.Mesh(noseGeom, bodyMat); nose.position.z = -3; // 放在机身前端 ship.add(nose); // ---- 主翼（一个扁平的盒子） ---- const wingGeom = new THREE.BoxGeometry(6, 0.1, 1.5); const wing = new THREE.Mesh(wingGeom, wingMat); wing.position.z = -0.5; ship.add(wing); // ---- 尾翼（垂直尾翼） ---- const tailGeom = new THREE.BoxGeometry(0.1, 1.5, 1); const tail = new THREE.Mesh(tailGeom, wingMat); tail.position.set(0, 1, 1.5); ship.add(tail); // ---- 引擎喷口（两个小圆柱） ---- const engineGeom = new THREE.CylinderGeometry(0.4, 0.5, 0.5, 8); engineGeom.rotateX(Math.PI / 2); const engineL = new THREE.Mesh(engineGeom, engineMat); engineL.position.set(-1, -0.2, 1.5); ship.add(engineL); const engineR = engineL.clone(); engineR.position.x = 1; ship.add(engineR); // ---- 轮廓线框（让飞船看起来是“轮廓”风格） ---- // 给机身和机翼添加边缘线 const bodyEdges = new THREE.EdgesGeometry(bodyGeom); const bodyLine = new THREE.LineSegments(bodyEdges, new THREE.LineBasicMaterial({ color: 0xffffff })); bodyLine.position.copy(body.position); ship.add(bodyLine); const noseEdges = new THREE.EdgesGeometry(noseGeom); const noseLine = new THREE.LineSegments(noseEdges, new THREE.LineBasicMaterial({ color: 0xffffff })); noseLine.position.copy(nose.position); ship.add(noseLine); const wingEdges = new THREE.EdgesGeometry(wingGeom); const wingLine = new THREE.LineSegments(wingEdges, new THREE.LineBasicMaterial({ color: 0xffffff })); wingLine.position.copy(wing.position); ship.add(wingLine); // 返回飞船组 return ship; }
